    private static byte[] compressAndSave(BufferedImage image, String outputPath, int targetSizeKb) throws IOException {
        float compressionQuality = 0.9f; // Valeur initiale de qualité

        // ByteArrayOutputStream pour stocker l'image compressée en mémoire
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        ByteArrayOutputStream outputStream = new ByteArrayOutputStream();
        // Réduire la qualité jusqu'à atteindre la taille cible ou une qualité minimale acceptable
        while (baos.size() > targetSizeKb * 1024 && compressionQuality > 0.01) {
            baos.reset();

            // Encoder l'image avec une qualité réduite
            ImageIO.write(image, "jpeg", baos);

            // Réduire la qualité pour la prochaine itération
            compressionQuality -= 0.5f;
        }
        //Sauvegarder l'image compressée sur le disque
        ImageIO.write(image, "jpeg", new File(outputPath));
        byte[] compressedImageData = baos.toByteArray();
        return compressedImageData;
    }
